📝 Ejemplos de Uso

Uso en Templates
<!-- Mostrar contenido basado en rol -->
<div *ngIf="roleService.hasRole('admin')">
  <button>Panel Administrativo</button>
</div>

<!-- Mostrar contenido basado en permiso -->
<button
  *ngIf="roleService.hasPermission('MANAGE_USERS')"
  (click)="manageUsers()">
  Gestionar Usuarios
</button>

<!-- Directiva personalizada -->
<div *hasPermission="'DELETE_CONTENT'">
  <button class="danger">Eliminar</button>
</div>
